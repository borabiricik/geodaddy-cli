```rust
/// Extract @type values from a JSON-LD value, handling @graph arrays.
fn extract_types(val: &Value, types: &mut HashSet<String>) {
    match val.get("@type") {
        Some(Value::String(s)) => { types.insert(s.clone()); }
        Some(Value::Array(arr)) => { for item in arr { if let Some(s) = item.as_str() { types.insert(s.to_string()); } } }
        _ => {}
    }
    if let Some(Value::Array(graph)) = val.get("@graph") {
        for item in graph { extract_types(item, types); }
    }
}
```

Note: extract_types is currently `fn` (private) in geo.rs. The executor should either:
(a) Make it `pub(crate)` in geo.rs and import it, OR
(b) Duplicate the helper in geo_entities.rs (simpler, no cross-module dependency)
Option (a) is preferred for DRY.

```rust
let ld_sel = Selector::parse(r#"script[type="application/ld+json"]"#).expect("valid selector");
let mut all_json: Vec<Value> = Vec::new();
let mut found_types: HashSet<String> = HashSet::new();
for block in html.select(&ld_sel) {

    if let Ok(val) = serde_json::from_str::<Value>(&text) {

        all_json.push(val);
    }
}
```
